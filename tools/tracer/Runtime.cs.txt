// 기록 런타임: 트레이서가 표본과 함께 컴파일하는 도우미 (표본과 같은 C# 9).
// 기록 호출 __T.L(줄, 변수...) 이 올 때마다 호출 스택(프레임별 변수), 힙(변수에서 닿는 객체), 새 콘솔 출력을 한 칸으로 남긴다.
using System;
using System.Collections;
using System.Collections.Generic;
using System.IO;
using System.Linq;
using System.Reflection;
using System.Runtime.CompilerServices;
using System.Text;
using System.Text.Json;

public readonly struct __Var
{
    public readonly string Name;
    public readonly object Value;
    /// 선언 형식이 참조 형식(object, 인터페이스)인데 값 타입이 들어 있다 = 박싱된 상자다
    public readonly bool Boxed;
    /// ref · in 매개변수 (부른 쪽 변수를 가리킨다)
    public readonly bool IsRef;
    /// 선언만 하고 값이 확실히 정해지지 않은 지역 변수. Value 는 형식 이름이다
    public readonly bool Unassigned;
    public __Var(string name, object value, bool boxed, bool isRef, bool unassigned = false) { Name = name; Value = value; Boxed = boxed; IsRef = isRef; Unassigned = unassigned; }
}

/// Console.ReadLine 이 읽는 입력. 읽은 줄을 콘솔 출력에 그대로 찍는다 (사람이 친 글자가 화면에 남는 것처럼)
public sealed class __EchoReader : TextReader
{
    readonly StringReader inner;
    public __EchoReader(string text) { inner = new StringReader(text); }
    public override int Peek() => inner.Peek();
    public override int Read() => inner.Read();
    public override string ReadLine()
    {
        string line = inner.ReadLine();
        if (line != null) Console.Out.Write(line + "\n");
        return line;
    }
}

public sealed class __GalleryTraceLimit : Exception
{
    public __GalleryTraceLimit() : base("기록이 너무 길다 (무한 반복이거나 표본이 너무 길다)") { }
}

public static class __T
{
    public const int MaxSteps = 1500;
    const int MaxItems = 40;

    sealed class Frame
    {
        public string Name;
        public __Var[] Vars = new __Var[0];
        /// ref 매개변수 이름 → 부른 쪽 (프레임 번호, 변수 이름). 프레임 번호 -1 은 정적 영역
        public readonly Dictionary<string, (int frame, string name)> Refs = new Dictionary<string, (int, string)>();
    }

    static readonly List<Frame> frames = new List<Frame>();

    /// 부르기 직전에 남긴 ref 인자 (메서드, 매개변수, 부른 쪽 프레임, 부른 쪽 변수). Enter 가 가져간다
    static readonly List<(string method, string param, int depth, string target)> pendingRefs = new List<(string, string, int, string)>();
    static readonly List<Dictionary<string, object>> steps = new List<Dictionary<string, object>>();
    static StringWriter output;
    static int outputPos;

    public static void Begin(string stdin)
    {
        output = new StringWriter { NewLine = "\n" };
        Console.SetOut(output);
        Console.SetIn(new __EchoReader(stdin ?? ""));
    }

    /// 선언만 하고 값이 확실히 정해지지 않은 지역 변수 (읽을 수 없으니 형식 이름만 넘긴다)
    public static __Var U(string name, string type) => new __Var(name, type, false, false, true);

    public static __Var V<T>(string name, T value) => new __Var(name, value, IsBoxed(typeof(T), value), false);

    /// ref · in 매개변수
    public static __Var VR<T>(string name, T value) => new __Var(name, value, IsBoxed(typeof(T), value), true);

    /// 참조 형식 자리(object, 인터페이스)에 값 타입이 들어 있으면 박싱된 상자다
    static bool IsBoxed(Type declared, object v) => v != null && !declared.IsValueType && v.GetType().IsValueType;

    /// 이 문장이 부를 메서드의 ref 인자. target 은 부른 쪽 변수 이름, "static:형식.필드", 모르면 ""
    public static void RefArg(string method, string param, string target) => pendingRefs.Add((method, param, frames.Count - 1, target));

    public static void Enter(string name)
    {
        var f = new Frame { Name = name };
        int caller = frames.Count - 1;
        for (int i = pendingRefs.Count - 1; i >= 0; i--)
        {
            var p = pendingRefs[i];
            if (p.method != name || p.depth != caller) continue;
            if (!f.Refs.ContainsKey(p.param))
            {
                if (p.target.StartsWith("static:")) f.Refs[p.param] = (-1, p.target.Substring(7));
                else if (p.target.Length > 0) f.Refs[p.param] = (caller, p.target);
            }
            pendingRefs.RemoveAt(i);
        }
        frames.Add(f);
    }

    public static void Exit()
    {
        if (frames.Count > 0) frames.RemoveAt(frames.Count - 1);
    }

    public static void L(int line, params __Var[] vars)
    {
        // 이 프레임(과 더 깊은 프레임)이 앞 문장에서 남긴 ref 인자는 이제 쓰이지 않는다
        pendingRefs.RemoveAll(p => p.depth >= frames.Count - 1);
        if (frames.Count > 0) frames[frames.Count - 1].Vars = vars;
        steps.Add(Snap(line));
        if (steps.Count >= MaxSteps) throw new __GalleryTraceLimit();
    }

    /// 트레이서가 실행 뒤에 부른다. error 는 잡히지 않은 예외 (없으면 null)
    public static string Result(string error, bool limit)
    {
        var end = new Dictionary<string, object> { ["line"] = steps.Count > 0 ? steps[steps.Count - 1]["line"] : 0, ["end"] = true };
        string rest = TakeOutput();
        if (rest.Length > 0) end["out"] = rest;
        if (error != null && steps.Count > 0)
        {
            // 예외로 끝나면 마지막 상태(스택, 힙)를 그대로 보여 준다
            var last = steps[steps.Count - 1];
            end["frames"] = last["frames"];
            end["heap"] = last["heap"];
            end["error"] = error;
        }
        else if (finalState != null || steps.Count > 0)
        {
            // 정상으로 끝나도 마지막 변수 상태를 남겨 둔다 (수업에서 최종 값을 메모리로 보이게).
            // 최상위 문은 마지막 문장 뒤 상태(Final), 그 밖(return 으로 일찍 끝남, Main 메서드)은 마지막 칸의 상태
            var last = finalState ?? steps[steps.Count - 1];
            end["frames"] = last["frames"];
            end["heap"] = last["heap"];
        }
        else
        {
            end["frames"] = new object[0];
            end["heap"] = StaticsOnlyHeap();
        }
        steps.Add(end);
        var result = new Dictionary<string, object> { ["steps"] = steps };
        if (error != null) result["error"] = error;
        if (limit) result["limit"] = true;
        return JsonSerializer.Serialize(result, new JsonSerializerOptions { Encoder = System.Text.Encodings.Web.JavaScriptEncoder.UnsafeRelaxedJsonEscaping });
    }

    static string TakeOutput()
    {
        if (output == null) return "";
        string all = output.ToString();
        string delta = all.Substring(outputPos);
        outputPos = all.Length;
        return delta;
    }

    // ───────── 한 칸 ─────────

    // 객체 번호는 처음 본 순서로 한 번 주고 끝까지 같다 (칸이 바뀌어도 같은 객체 = 같은 번호. 무대의 바뀐 값 표시와 비유 장면의 움직임이 쓴다).
    // 약한 표라 기록이 객체의 수명을 늘리지 않는다
    sealed class IdBox { public int Id; }
    static readonly ConditionalWeakTable<object, IdBox> ids = new ConditionalWeakTable<object, IdBox>();
    static int nextId;

    static int IdOf(object v) => ids.GetValue(v, _ => new IdBox { Id = ++nextId }).Id;

    sealed class Snapshot
    {
        public readonly Dictionary<object, int> Ids = new Dictionary<object, int>(ReferenceEqualityComparer.Instance);
        public readonly Queue<object> Pending = new Queue<object>();
        public readonly List<object> Heap = new List<object>();
    }

    static Dictionary<string, object> finalState;

    /// 최상위 문의 마지막 문장 뒤 변수 상태. 프로그램 끝 칸이 이것을 보인다 (출력은 끝 칸이 가져간다)
    public static void Final(params __Var[] vars)
    {
        if (frames.Count > 0) frames[frames.Count - 1].Vars = vars;
        finalState = Snap(0, false);
    }

    static Dictionary<string, object> Snap(int line, bool takeOutput = true)
    {
        var s = new Snapshot();
        var frameList = new List<object>();
        var statics = Statics(s);
        int offset = statics.Count > 0 ? 1 : 0;
        if (statics.Count > 0) frameList.Add(new Dictionary<string, object> { ["name"] = "static", ["vars"] = statics });
        var varLists = new List<List<object[]>>();
        foreach (var f in frames)
        {
            var vars = new List<object[]>();
            foreach (var v in f.Vars)
            {
                object val = v.Unassigned
                    ? new Dictionary<string, object> { ["k"] = "x", ["t"] = (string)v.Value }
                    : Value(v.Value, s, 0, v.Boxed);
                if (v.IsRef)
                {
                    // ref 매개변수: 값과 함께 부른 쪽 변수의 자리(무대의 프레임 번호, 이름)를 남긴다
                    var r = new Dictionary<string, object> { ["k"] = "ref", ["v"] = val };
                    if (f.Refs.TryGetValue(v.Name, out var to))
                        r["to"] = new object[] { to.frame < 0 ? 0 : to.frame + offset, to.name };
                    val = r;
                }
                vars.Add(new object[] { v.Name, val });
            }
            varLists.Add(vars);
            frameList.Add(new Dictionary<string, object> { ["name"] = f.Name, ["vars"] = vars });
        }
        // 부른 쪽 변수는 그 프레임이 마지막으로 기록한 값이라, ref 로 고친 값이 아직 없다. ref 매개변수의 지금 값으로 바꿔 둔다 (같은 자리다).
        // 위 프레임부터 내려가야 ref 를 다시 ref 로 넘긴 사슬도 이어진다
        for (int i = frames.Count - 1; i >= 0; i--)
            foreach (var pair in varLists[i])
            {
                if (!(pair[1] is Dictionary<string, object> r) || !r.ContainsKey("to")) continue;
                if (!frames[i].Refs.TryGetValue((string)pair[0], out var to) || to.frame < 0 || to.frame >= i) continue;
                foreach (var target in varLists[to.frame])
                {
                    if ((string)target[0] != to.name) continue;
                    if (target[1] is Dictionary<string, object> tr && (string)tr["k"] == "ref") tr["v"] = r["v"];
                    else target[1] = r["v"];
                }
            }
        Drain(s);
        var step = new Dictionary<string, object> { ["line"] = line, ["frames"] = frameList, ["heap"] = s.Heap };
        string delta = takeOutput ? TakeOutput() : "";
        if (delta.Length > 0) step["out"] = delta;
        return step;
    }

    static List<object> StaticsOnlyHeap()
    {
        var s = new Snapshot();
        Statics(s);
        Drain(s);
        return s.Heap;
    }

    static void Drain(Snapshot s)
    {
        while (s.Pending.Count > 0) s.Heap.Add(HeapObject(s.Pending.Dequeue(), s));
    }

    static IEnumerable<Type> UserTypes() =>
        typeof(__T).Assembly.GetTypes().Where(t => !t.Name.StartsWith("__") && !t.Name.Contains("<") && !t.IsGenericTypeDefinition);

    static List<object> Statics(Snapshot s)
    {
        var list = new List<object>();
        foreach (var t in UserTypes())
            foreach (var f in t.GetFields(BindingFlags.Static | BindingFlags.Public | BindingFlags.NonPublic | BindingFlags.DeclaredOnly))
            {
                if (f.IsLiteral || f.Name.Contains("<")) continue;
                object v = f.GetValue(null);
                list.Add(new object[] { $"{t.Name}.{f.Name}", Value(v, s, 0, IsBoxed(f.FieldType, v)) });
            }
        return list;
    }

    // ───────── 값 ─────────

    static readonly Dictionary<Type, string> Names = new Dictionary<Type, string>
    {
        [typeof(int)] = "int", [typeof(long)] = "long", [typeof(short)] = "short", [typeof(byte)] = "byte", [typeof(sbyte)] = "sbyte",
        [typeof(uint)] = "uint", [typeof(ulong)] = "ulong", [typeof(ushort)] = "ushort", [typeof(float)] = "float", [typeof(double)] = "double",
        [typeof(decimal)] = "decimal", [typeof(bool)] = "bool", [typeof(char)] = "char", [typeof(string)] = "string", [typeof(object)] = "object",
    };

    public static string TypeName(Type t)
    {
        if (Names.TryGetValue(t, out var n)) return n;
        if (t.IsArray) return TypeName(t.GetElementType()) + "[" + new string(',', t.GetArrayRank() - 1) + "]";
        if (t.IsGenericType)
        {
            string name = t.Name.Split('`')[0];
            if (name == "Nullable") return TypeName(t.GetGenericArguments()[0]) + "?";
            return name + "<" + string.Join(", ", t.GetGenericArguments().Select(TypeName)) + ">";
        }
        return t.Name;
    }

    static bool IsScalar(Type t) => t.IsPrimitive || t.IsEnum || t == typeof(string) || t == typeof(decimal);

    static string Scalar(object v)
    {
        switch (v)
        {
            case bool b: return b ? "true" : "false";
            case float f: return f.ToString("R", System.Globalization.CultureInfo.InvariantCulture);
            case double d: return d.ToString("R", System.Globalization.CultureInfo.InvariantCulture);
            case IFormattable x: return x.ToString(null, System.Globalization.CultureInfo.InvariantCulture);
            default: return v.ToString();
        }
    }

    /// boxed: 참조 형식 자리에 든 값 타입이다. 값이 아니라 힙의 상자를 가리키는 참조로 그린다
    static object Value(object v, Snapshot s, int depth, bool boxed = false)
    {
        if (v == null) return new Dictionary<string, object> { ["k"] = "n" };
        var t = v.GetType();
        if (boxed) return Ref(v, t, s);
        if (IsScalar(t)) return new Dictionary<string, object> { ["k"] = "v", ["t"] = TypeName(t), ["v"] = t.IsEnum ? t.Name + "." + v : Scalar(v) };
        if (t.IsValueType)
        {
            if (t.Namespace != null && t.Namespace.StartsWith("System") || depth > 3)
                return new Dictionary<string, object> { ["k"] = "v", ["t"] = TypeName(t), ["v"] = Scalar(v) };
            return new Dictionary<string, object> { ["k"] = "s", ["t"] = TypeName(t), ["f"] = Fields(v, t, s, depth + 1) };
        }
        return Ref(v, t, s);
    }

    static object Ref(object v, Type t, Snapshot s)
    {
        if (!s.Ids.TryGetValue(v, out int id))
        {
            id = IdOf(v);
            s.Ids[v] = id;
            s.Pending.Enqueue(v);
        }
        return new Dictionary<string, object> { ["k"] = "r", ["t"] = TypeName(t), ["id"] = id };
    }

    static string FieldName(FieldInfo f)
    {
        // 자동 속성의 숨은 필드 <Name>k__BackingField 는 속성 이름으로 보인다
        if (f.Name.StartsWith("<") && f.Name.EndsWith(">k__BackingField")) return f.Name.Substring(1, f.Name.IndexOf('>') - 1);
        return f.Name;
    }

    static List<object> Fields(object v, Type t, Snapshot s, int depth)
    {
        var list = new List<object>();
        var chain = new List<Type>();
        for (var c = t; c != null && c != typeof(object) && c != typeof(ValueType); c = c.BaseType) chain.Insert(0, c);
        foreach (var c in chain)
            foreach (var f in c.GetFields(BindingFlags.Instance | BindingFlags.Public | BindingFlags.NonPublic | BindingFlags.DeclaredOnly))
            {
                if (f.Name.StartsWith("<") && !f.Name.EndsWith(">k__BackingField")) continue;
                object x = f.GetValue(v);
                list.Add(new object[] { FieldName(f), Value(x, s, depth, IsBoxed(f.FieldType, x)) });
            }
        return list;
    }

    static object HeapObject(object v, Snapshot s)
    {
        var t = v.GetType();
        var o = new Dictionary<string, object> { ["id"] = s.Ids[v], ["t"] = TypeName(t) };
        if (t.IsValueType)
        {
            // 박싱된 값: 힙의 상자 안에 값 하나(또는 구조체 필드)가 들어 있다
            o["kind"] = "box";
            bool plain = IsScalar(t) || t.Namespace != null && t.Namespace.StartsWith("System");
            o["f"] = plain ? new List<object>() : Fields(v, t, s, 1);
            if (plain) o["text"] = t.IsEnum ? t.Name + "." + v : Scalar(v);
            return o;
        }
        if (v is Delegate d)
        {
            o["kind"] = "fn";
            o["f"] = new List<object>();
            o["text"] = d.Method.Name.Contains("<") ? "람다" : d.Method.Name;
            return o;
        }
        if (t.IsArray)
        {
            var a = (Array)v;
            var elem = t.GetElementType();
            o["kind"] = "arr";
            var items = new List<object>();
            if (a.Rank == 1)
            {
                for (int i = 0; i < a.Length && i < MaxItems; i++) items.Add(new object[] { i.ToString(), Item(a.GetValue(i), elem, s) });
            }
            else
            {
                int n = 0;
                foreach (var idx in Indices(a))
                {
                    if (n++ >= MaxItems) break;
                    items.Add(new object[] { string.Join(",", idx), Item(a.GetValue(idx), elem, s) });
                }
            }
            o["f"] = items;
            if (a.Length > MaxItems) o["more"] = a.Length - MaxItems;
            return o;
        }
        bool system = t.Namespace != null && t.Namespace.StartsWith("System");
        // 컬렉션은 System.Collections(.Generic) 것만 펼친다. LINQ 결과처럼 늦게 계산되는 것을 펼치면 표본 코드가 더 돈다
        bool collection = t.Namespace == "System.Collections.Generic" || t.Namespace == "System.Collections";
        if (collection && v is IDictionary dict)
        {
            o["kind"] = "dict";
            var items = new List<object>();
            foreach (DictionaryEntry e in dict)
            {
                if (items.Count >= MaxItems) break;
                items.Add(new object[] { e.Key is string k ? "\"" + k + "\"" : Scalar(e.Key), Item(e.Value, ElementType(t, 1), s) });
            }
            o["f"] = items;
            return o;
        }
        if (collection && v is IEnumerable)
        {
            o["kind"] = v is IList ? "list" : "seq";
            var items = new List<object>();
            int i = 0;
            foreach (var x in (IEnumerable)v)
            {
                if (i >= MaxItems) break;
                items.Add(new object[] { (i++).ToString(), Item(x, ElementType(t, 0), s) });
            }
            o["f"] = items;
            return o;
        }
        o["kind"] = "obj";
        if (system)
        {
            // StringBuilder 같은 시스템 객체는 속을 그리지 않고 ToString 이 의미 있을 때만 글로 보인다
            o["f"] = new List<object>();
            var ts = t.GetMethod("ToString", Type.EmptyTypes);
            if (ts != null && ts.DeclaringType != typeof(object)) o["text"] = v.ToString();
            return o;
        }
        o["f"] = Fields(v, t, s, 0);
        return o;
    }

    /// 배열·컬렉션 칸. 칸 형식이 object 같은 참조 형식인데 값 타입이 들어 있으면 박싱된 상자다
    static object Item(object x, Type declared, Snapshot s) => Value(x, s, 0, IsBoxed(declared, x));

    /// 제네릭 컬렉션의 칸 형식 (List<T> 는 T, Dictionary<K, V> 의 값은 V). 비제네릭은 object
    static Type ElementType(Type t, int arg)
    {
        var args = t.IsGenericType ? t.GetGenericArguments() : Type.EmptyTypes;
        return arg < args.Length ? args[arg] : typeof(object);
    }

    static IEnumerable<int[]> Indices(Array a)
    {
        var idx = new int[a.Rank];
        for (int n = 0; n < a.Length; n++)
        {
            yield return (int[])idx.Clone();
            for (int r = a.Rank - 1; r >= 0; r--)
            {
                if (++idx[r] < a.GetLength(r)) break;
                idx[r] = 0;
            }
        }
    }
}
